# DriftLab monitor — cron template (any scheduler works; the command is idempotent).
# Daily at 07:00. The staleness policy decides which agents are actually due.
# Replace /absolute/path/to/driftlab with the repository's absolute path.
0 7 * * * cd /absolute/path/to/driftlab && /usr/bin/env driftlab monitor run --scheduled >> ~/.driftlab/monitor.log 2>&1
